async function cargarReporte() {
  const cuerpoTabla = document.getElementById('tabla-contactos');
  const respuesta = await fetch('/api/contactos');
  const resultado = await respuesta.json();

  if (!respuesta.ok) {
    throw new Error(resultado.mensaje || 'No se pudo cargar el reporte.');
  }

  if (!Array.isArray(resultado.contactos)) {
    throw new Error('El servidor devolvió un reporte no válido.');
  }

  cuerpoTabla.replaceChildren();

  if (resultado.contactos.length === 0) {
    const fila = document.createElement('tr');
    const celda = document.createElement('td');
    celda.colSpan = 7;
    celda.textContent = 'No hay contactos registrados.';

    cuerpoTabla.appendChild(fila);
    mostrarMensaje('No hay contactos registrados.', 'info');

  const campos = [
    'id',
    'cedula',
    'nombre',

async function cargarContactosParaBorrar() {
  const cuerpoTabla = document.getElementById('tabla-borrar-contactos');
  const respuesta = await fetch('/api/contactos');
  const resultado = await respuesta.json();

  if (!respuesta.ok) {
    throw new Error(resultado.mensaje || 'No se pudieron cargar los contactos.');
  }

  if (!Array.isArray(resultado.contactos)) {
    throw new Error('El servidor devolvió una lista de contactos no válida.');
  }

  cuerpoTabla.replaceChildren();

  if (resultado.contactos.length === 0) {
    mostrarTablaBorrarVacia(cuerpoTabla);
    mostrarMensaje('No hay contactos registrados.', 'info');
    return;
  }

  const campos = [
    'id',
    'cedula',
    'nombre',
    'apellido',
    'telefono',
    'correo_electronico',
    'direccion'
  ];

  for (const contacto of resultado.contactos) {
    const fila = document.createElement('tr');
    fila.dataset.contactoId = contacto.id;

    for (const campo of campos) {
      const celda = document.createElement('td');
      celda.textContent = contacto[campo] ?? '';
      fila.appendChild(celda);
    }

    const celdaAccion = document.createElement('td');
    const boton = document.createElement('button');
    boton.type = 'button';
    boton.className = 'btn btn-danger btn-sm';
    boton.textContent = 'Borrar';
    boton.dataset.borrarId = contacto.id;
    boton.dataset.cedula = contacto.cedula ?? '';
    boton.setAttribute('aria-label', `Borrar contacto con cédula ${contacto.cedula ?? ''}`);

    celdaAccion.appendChild(boton);
    fila.appendChild(celdaAccion);
    cuerpoTabla.appendChild(fila);
  }

  mostrarMensaje(`Contactos cargados: ${resultado.contactos.length}.`, 'info');
}

### 4. Enviar la eliminación desde `app.js`

Agrega un manejador de clics delegado. La confirmación evita borrar accidentalmente; si el servidor responde con error, la fila se conserva.

````javascript
document.addEventListener('click', async (evento) => {
  const boton = evento.target.closest('[data-borrar-id]');
  if (!boton) return;

  const id = boton.dataset.borrarId;
  const cedula = boton.dataset.cedula;

  if (!window.confirm(`¿Deseas borrar el contacto con cédula ${cedula}?`)) {
    return;
  }

  boton.disabled = true;

  try {
    const respuesta = await fetch(`/api/contactos/${encodeURIComponent(id)}`, {
      method: 'DELETE'
    });
    const resultado = await respuesta.json();

    if (!respuesta.ok) {
      throw new Error(resultado.mensaje || 'No se pudo borrar el contacto.');
    }

    const fila = boton.closest('tr');
    const cuerpoTabla = document.getElementById('tabla-borrar-contactos');
    fila.remove();

    if (cuerpoTabla.querySelectorAll('tr[data-contacto-id]').length === 0) {
      mostrarTablaBorrarVacia(cuerpoTabla);
    }

    mostrarMensaje(resultado.mensaje, 'success');
  } catch (error) {
    mostrarMensaje(error.message || 'No se pudo contactar al servidor.', 'danger');
  } finally {
    if (boton.isConnected) boton.disabled = false;
  }
});
````

````javascript
app.delete('/api/contactos/:id', async (req, res) => {
  const id = req.params.id;

  if (!/^[1-9]\d*$/.test(id) || Number(id) > 2147483647) {
    return res.status(400).json({ mensaje: 'El ID del contacto no es válido.' });
  }

  try {
    const resultado = await pool.query(
      `DELETE FROM contactos
       WHERE id = $1
       RETURNING id, cedula`,
      [Number(id)]
    );

    if (resultado.rowCount === 0) {
      return res.status(404).json({
        mensaje: 'El contacto ya no existe o no fue encontrado.'
      });
    }

    return res.json({
      mensaje: `Se borró el contacto con cédula ${resultado.rows[0].cedula}.`
    });
  } catch (error) {
    console.error('Error al borrar el contacto:', error.message);
    return res.status(500).json({
      mensaje: 'No se pudo borrar el contacto.'
    });
  }
});
````

### Flujo

Al seleccionar **Borrar**, `app.js` carga la vista y pide los contactos al servidor. Cada fila incluye un botón asociado al ID del registro. Al confirmar, el navegador envía el DELETE; si PostgreSQL confirma la eliminación, se quita la fila y se informa el resultado en `#mensajes`.


DESCRIPCIÓN DE ACCIÓN DE BORRAR (DELETE)

`app.delete()` de Express registra una ruta para atender solicitudes HTTP **DELETE**. En este caso, la ruta `'/api/contactos/:id'` permite solicitar la eliminación de un contacto según su ID.

El flujo del manejador es:

1. **Obtiene el ID** desde `req.params.id`. Por ejemplo, al solicitar `DELETE /api/contactos/15`, el valor es `"15"`.
2. **Valida el formato y rango**: exige un entero positivo que quepa en la columna `id` de PostgreSQL. Si no es válido, responde con estado `400`.
3. **Elimina el registro** con una consulta SQL parametrizada. `RETURNING` permite saber si se borró una fila y recuperar sus datos para el mensaje.
4. **Responde según el resultado**:
   - Si no se encontró ningún contacto, devuelve `404`.
   - Si se eliminó, devuelve `200` con un mensaje de confirmación.
   - Si falla la operación de base de datos, devuelve `500`.

El navegador envía esa solicitud cuando el usuario confirma el borrado. Se usa `DELETE` porque expresa que se quiere eliminar el recurso, en vez de actualizarlo o crear uno nuevo.

### 4. Enviar cambios desde `app.js`

Agrega un manejador de envío para el formulario de actualización. El objeto enviado contiene únicamente campos editables:

````javascript
document.addEventListener('submit', async (evento) => {
  if (evento.target.id !== 'form-actualizar-contacto') return;

  evento.preventDefault();

  const formulario = evento.target;
  const boton = formulario.querySelector('[type="submit"]');
  const id = formulario.dataset.contactoId;
  boton.disabled = true;

  const cambios = {
    nombre: formulario.elements.nombre.value.trim(),
    apellido: formulario.elements.apellido.value.trim(),
    telefono: formulario.elements.telefono.value.trim(),
    direccion: formulario.elements.direccion.value.trim()
  };

  try {
    const respuesta = await fetch(`/api/contactos/${encodeURIComponent(id)}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(cambios)
    });

    const resultado = await respuesta.json();

    if (!respuesta.ok) {
      throw new Error(resultado.mensaje || 'No se pudo actualizar el contacto.');
    }

    await mostrarVista('actualizar.html');
    await cargarContactosParaActualizar();
    mostrarMensaje(resultado.mensaje || 'Contacto actualizado correctamente.', 'success');
  } catch (error) {
    mostrarMensaje(error.message || 'No se pudo contactar al servidor.', 'danger');
  } finally {
    if (boton.isConnected) boton.disabled = false;
  }
});
````

### 5. Ruta PATCH en `server.js`

Añade esta ruta junto a las demás rutas de la API. No modifica cédula ni correo electrónico, incluso si alguien intenta incluirlos manualmente en la solicitud.

````javascript
app.patch('/api/contactos/:id', async (req, res) => {
  const id = req.params.id;

  if (!/^[1-9]\d*$/.test(id) || Number(id) > 2147483647) {
    return res.status(400).json({ mensaje: 'El ID del contacto no es válido.' });
  }

  const campos = ['nombre', 'apellido', 'telefono', 'direccion'];
  const cambios = {};

  for (const campo of campos) {
    const valor = req.body?.[campo];

    if (typeof valor !== 'string' || !valor.trim()) {
      return res.status(400).json({
        mensaje: `El campo ${campo} es obligatorio.`
      });
    }

    cambios[campo] = valor.trim();
  }

  const limites = {
    nombre: 80,
    apellido: 80,
    telefono: 15
  };

  for (const [campo, limite] of Object.entries(limites)) {
    if (cambios[campo].length > limite) {
      return res.status(400).json({
        mensaje: `El campo ${campo} no puede superar ${limite} caracteres.`
      });
    }
  }

  try {
    const resultado = await pool.query(
      `UPDATE contactos
       SET nombre = $1,
           apellido = $2,
           telefono = $3,
           direccion = $4
       WHERE id = $5
       RETURNING id`,
      [
        cambios.nombre,
        cambios.apellido,
        cambios.telefono,
        cambios.direccion,
        Number(id)
      ]
    );

    if (resultado.rowCount === 0) {
      return res.status(404).json({
        mensaje: 'No se encontró el contacto que se desea actualizar.'
      });
    }

    return res.json({
      mensaje: 'Contacto actualizado correctamente.'
    });
  } catch (error) {
    console.error('Error al actualizar el contacto:', error.message);
    return res.status(500).json({
      mensaje: 'No se pudo actualizar el contacto.'
    });
  }
});
````

**Flujo:** al elegir **Actualizar**, se muestra la tabla; el botón de una fila abre su formulario; **Guardar** envía los cuatro campos permitidos mediante PATCH; al tener éxito, se vuelve al listado y se consulta nuevamente la tabla para mostrar los cambios.
